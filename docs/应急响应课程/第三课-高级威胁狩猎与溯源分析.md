第三课  高级威胁狩猎与溯源分析
APT攻击与ATT&CK框架
APT（高级持续性威胁）攻击通常遵循多阶段攻击链：侦察→武器化→投递→漏洞利用→安装→指挥控制→目标达成。MITRE ATT&CK框架将攻击行为映射到12个战术阶段（TA0001-TA0040），每个战术包含多种技术。通过将检测到的行为映射到ATT&CK技术，可以重建攻击链并识别遗漏阶段。
威胁情报集成
威胁情报（TI）提供已知IOC（IP、域名、文件哈希）和TTP（战术、技术和过程）。通过API（如MISP）自动获取情报，与内部日志关联，可快速识别已知威胁。STIX/TAXII是威胁情报共享标准。
网络流量分析
基于PCAP或NetFlow分析网络流量，提取五元组、流量特征、DNS请求等。异常检测包括：DGA域名检测、Beacon检测、端口扫描识别等。可使用Scapy解析数据包，或使用pyshark、dpkt。
异常检测算法
基于统计基线的异常检测（如标准差、百分位数）和机器学习方法（如孤立森林）可用于识别偏离正常的行为模式。例如，登录失败次数突增、流量体积异常、进程创建模式异常。
端点检测与响应（EDR）
在主机上部署轻量级传感器，监控进程创建、文件变化、网络连接、注册表修改等行为。通过实时监控和规则匹配，检测恶意活动并触发告警。
攻击溯源技术
通过关联日志（主机、网络、应用）构建时间线，使用图数据库（如Neo4j）或NetworkX重建攻击路径，确定初始入侵点、横向移动路径和数据外泄通道。
报告生成与证据固定
自动化生成包含时间线、IOC、受影响资产、处置建议的报告，并打包关键证据（内存转储、日志片段、网络流量）供后续分析。

"""
attack_matrix_mapper.py
ATT&CK攻击链映射器
"""
import json
from datetime import datetime
from collections import defaultdict

class AttackMapper:
    """
    将检测到的告警映射到MITRE ATT&CK战术
    """
    def __init__(self):
        # 简化的ATT&CK技术到战术映射（真实环境需加载完整JSON）
        self.technique_to_tactic = {
            "T1566": "TA0001",  # 鱼叉式钓鱼链接 -> 初始访问
            "T1059": "TA0002",  # 命令行执行 -> 执行
            "T1547": "TA0003",  # 启动项持久化 -> 持久化
            "T1068": "TA0004",  # 权限提升漏洞 -> 权限提升
            "T1027": "TA0005",  # 混淆文件 -> 防御规避
            "T1003": "TA0006",  # 凭据转储 -> 凭据访问
            "T1082": "TA0007",  # 系统信息发现 -> 发现
            "T1021": "TA0008",  # 远程服务横向移动 -> 横向移动
            "T1071": "TA0010",  # 应用层协议C2 -> 命令控制
            "T1048": "TA0011",  # 数据外传 -> 数据渗出
        }
        self.tactic_names = {
            "TA0001": "初始访问", "TA0002": "执行", "TA0003": "持久化",
            "TA0004": "权限提升", "TA0005": "防御规避", "TA0006": "凭据访问",
            "TA0007": "发现", "TA0008": "横向移动", "TA0009": "收集",
            "TA0010": "命令控制", "TA0011": "数据渗出", "TA0040": "影响"
        }
        self.alerts = []
        self.attack_chain = defaultdict(list)

    def add_alert(self, timestamp, technique_id, description, source):
        """添加告警"""
        tactic_id = self.technique_to_tactic.get(technique_id)
        if not tactic_id:
            tactic_id = "TA0000"  # 未知
        alert = {
            "timestamp": datetime.fromisoformat(timestamp) if isinstance(timestamp, str) else timestamp,
            "technique_id": technique_id,
            "tactic_id": tactic_id,
            "description": description,
            "source": source
        }
        self.alerts.append(alert)
        self.attack_chain[tactic_id].append(alert)

    def reconstruct_chain(self):
        """按时间排序，重建攻击链"""
        all_alerts = sorted(self.alerts, key=lambda x: x["timestamp"])
        chain = []
        for alert in all_alerts:
            chain.append({
                "timestamp": alert["timestamp"],
                "tactic": self.tactic_names.get(alert["tactic_id"], alert["tactic_id"]),
                "technique": alert["technique_id"],
                "description": alert["description"],
                "source": alert["source"]
            })
        return chain

    def generate_report(self):
        """生成报告"""
        chain = self.reconstruct_chain()
        print("\n" + "="*60)
        print("攻击链时间线")
        print("="*60)
        for step in chain:
            print(f"{step['timestamp']} [{step['tactic']}] {step['technique']} - {step['description']}")

        # 检测缺失战术
        detected_tactics = set(self.attack_chain.keys())
        all_tactics = set(self.tactic_names.keys())
        missing = all_tactics - detected_tactics
        if missing:
            print("\n[!] 可能遗漏的战术阶段:")
            for tactic in missing:
                print(f"  - {self.tactic_names.get(tactic, tactic)}")

if __name__ == "__main__":

练习3-3：网络流量分析（PCAP解析）
目标：解析PCAP文件，提取HTTP请求、DNS查询，检测可疑域名。

"""
pcap_analyzer.py
PCAP文件分析器（使用scapy）
需要安装：pip install scapy
"""
from scapy.all import rdpcap, IP, TCP, UDP, DNS, Raw
from collections import Counter

class PCAPAnalyzer:
    def __init__(self, pcap_file):
        self.pcap_file = pcap_file
        self.packets = rdpcap(pcap_file)
        self.http_requests = []
        self.dns_queries = []
        self.suspicious_domains = []

    def extract_http_requests(self):
        """提取HTTP请求中的Host和URI"""
        for pkt in self.packets:
            if IP in pkt and TCP in pkt:
                if pkt[TCP].dport == 80 or pkt[TCP].sport == 80:
                    if Raw in pkt:
                        payload = pkt[Raw].load.decode(errors='ignore')
                        if 'GET' in payload or 'POST' in payload:
                            # 简单提取Host和请求行
                            lines = payload.split('\r\n')
                            for line in lines:
                                if line.startswith(('GET', 'POST', 'Host:')):
                                    self.http_requests.append(line)
        return self.http_requests

    def extract_dns_queries(self):
        """提取DNS查询域名"""
        for pkt in self.packets:
            if DNS in pkt and pkt[DNS].qr == 0:  # 查询
                qname = pkt[DNS].qd.qname.decode('utf-8').rstrip('.')
                self.dns_queries.append(qname)
        return self.dns_queries

    def detect_dga(self, threshold=10):
        """简单DGA检测：熵值过高或子域名过多"""
        from collections import defaultdict
        domain_parts = defaultdict(list)
        for domain in self.dns_queries:
            parts = domain.split('.')
            if len(parts) > 3:
                sub = '.'.join(parts[:-2])
                domain_parts[sub].append(domain)
        for sub, domains in domain_parts.items():
            if len(domains) > threshold:
                self.suspicious_domains.append({
                    "subdomain": sub,
                    "count": len(domains),
                    "samples": domains[:5]
                })
        return self.suspicious_domains

    def generate_report(self):
        print(f"分析PCAP文件: {self.pcap_file}")
        print(f"总包数: {len(self.packets)}")
        print(f"HTTP请求示例: {self.http_requests[:3]}")
        print(f"DNS查询示例: {self.dns_queries[:5]}")
        if self.suspicious_domains:
            print("\n可疑DGA域名:")
            for d in self.suspicious_domains:
                print(f"  {d['subdomain']} -> {d['count']}次查询, 示例: {d['samples']}")

"""
edr_sensor.py
EDR传感器（模拟，使用psutil）
需要安装：pip install psutil
"""
import psutil
import time
import hashlib
import json
from datetime import datetime

class EDREngine:
    def __init__(self):
        self.baseline_processes = set()
        self.alerts = []

    def capture_baseline(self):
        """捕获初始进程快照"""
        for proc in psutil.process_iter(['pid', 'name', 'exe']):
            self.baseline_processes.add(proc.info['pid'])
        print(f"基线进程数: {len(self.baseline_processes)}")

    def monitor_new_processes(self, interval=1):
        """监控新进程创建"""
        current_pids = set()
        while True:
            new_pids = set()
            for proc in psutil.process_iter(['pid', 'name', 'exe']):
                pid = proc.info['pid']
                current_pids.add(pid)
                if pid not in self.baseline_processes:
                    new_pids.add(pid)
            for pid in new_pids:
                try:
                    proc = psutil.Process(pid)
                    cmdline = ' '.join(proc.cmdline())
                    exe = proc.exe()
                    # 计算可执行文件MD5
                    try:
                        with open(exe, 'rb') as f:
                            md5 = hashlib.md5(f.read()).hexdigest()
                    except:
                        md5 = "N/A"
                    alert = {
                        "timestamp": datetime.now().isoformat(),
                        "type": "process_creation",
                        "pid": pid,
                        "name": proc.name(),
                        "cmdline": cmdline,
                        "exe": exe,
                        "md5": md5
                    }
                    self.alerts.append(alert)
                    self.check_ioc(alert)
                except (psutil.NoSuchProcess, psutil.AccessDenied):
                    pass
            self.baseline_processes = current_pids
            time.sleep(interval)

    def check_ioc(self, alert):
        """基于IOC检查（示例）"""
        suspicious_names = ["cmd.exe", "powershell.exe", "wscript.exe"]
        if alert['name'].lower() in suspicious_names:
            print(f"[!] 可疑进程: {alert['name']} PID={alert['pid']} CMD={alert['cmdline']}")

    def generate_report(self):
        with open('edr_alerts.json', 'w') as f:
            json.dump(self.alerts, f, indent=2)
        print(f"告警已保存到 edr_alerts.json，共 {len(self.alerts)} 条")

if __name__ == "__main__":
    edr = EDREngine()
    edr.capture_baseline()
    print("开始监控新进程...")
    try:
        edr.monitor_new_processes(interval=2)
    except KeyboardInterrupt:
        edr.generate_report()

使用psutil实时监控进程创建，记录详细信息（命令行、哈希）。
可扩展监控网络连接、文件变化、注册表修改。
将告警写入JSON文件，供后续分析。

练习3-5：攻击溯源与时间线重建
目标：从多个数据源（进程日志、网络连接、文件变化）构建攻击时间线。

"""
attack_tracer.py
攻击溯源时间线构建器
"""
from datetime import datetime
import json

class AttackTracer:
    def __init__(self):
        self.events = []

    def add_event(self, timestamp, source, event_type, details):
        """添加事件"""
        self.events.append({
            "timestamp": datetime.fromisoformat(timestamp) if isinstance(timestamp, str) else timestamp,
            "source": source,
            "type": event_type,
            "details": details
        })

    def build_timeline(self):
        """按时间排序"""
        self.events.sort(key=lambda x: x["timestamp"])
        return self.events

    def trace_attack(self):
        """简化的溯源：查找初始入侵点"""
        timeline = self.build_timeline()
        if not timeline:
            return
        # 假设初始入侵通常是用户执行恶意文件或钓鱼
        initial = None
        for e in timeline:
            if e["type"] == "process_creation" and "cmd.exe" in e["details"].get("cmdline", ""):
                initial = e
                break
            if e["type"] == "file_created" and ".exe" in e["details"].get("path", ""):
                initial = e
                break
        return initial

    def generate_report(self):
        timeline = self.build_timeline()
        print("\n攻击时间线")
        print("="*50)
        for e in timeline:
            print(f"{e['timestamp']} [{e['source']}] {e['type']} - {e['details']}")
        init = self.trace_attack()
        if init:
            print(f"\n可能初始入侵点: {init['timestamp']} {init['type']} {init['details']}")
