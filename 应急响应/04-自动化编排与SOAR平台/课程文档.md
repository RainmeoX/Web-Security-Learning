第四课：自动化编排与应急响应平台建设
知识点讲解
SOAR架构与工作流
SOAR（安全编排自动化与响应）将人员、流程和技术整合，通过自动化剧本（Playbook）定义事件响应流程。核心组件：触发器（告警）、编排引擎（条件判断、动作执行）、动作（调用API）、集成器（连接第三方工具）。
告警聚合与富化
从多个源（SIEM、EDR、防火墙）接收告警，进行去重、聚合（同一IP、同一用户），并富化上下文（地理位置、威胁情报、资产信息），减少告警疲劳。
自动化响应动作
常见响应动作：IP封堵（防火墙API）、主机隔离（云API或Ansible）、发送通知（邮件、Slack）、创建工单（ServiceNow、Jira）。动作应具备幂等性和错误重试机制。
剧本编排引擎
基于工作流引擎（如状态机、DAG）实现复杂逻辑：顺序执行、条件分支、并行处理、等待人工审批。可定义剧本文件（YAML/JSON）动态加载。
工单系统集成
通过REST API自动创建工单，将事件信息、处置建议、证据链接同步给运营团队。
应急响应度量
关键指标：平均检测时间（MTTD）、平均响应时间（MTTR）、告警处理量、误报率、自动化率。定期统计并生成报表。
自动化报告
使用模板引擎（Jinja2）生成HTML报告，或使用ReportLab生成PDF，通过邮件自动分发。

"""
soar_engine.py
SOAR编排引擎（简单版）
"""
import time
from datetime import datetime

class Action:
    """动作基类"""
    def execute(self, context):
        raise NotImplementedError

class BlockIPAction(Action):
    def execute(self, context):
        ip = context.get("ip")
        print(f"[{datetime.now()}] 封禁IP: {ip}")
        # 模拟调用防火墙API
        return {"status": "success", "ip": ip}

class NotifyAction(Action):
    def execute(self, context):
        message = context.get("message", "未提供消息")
        print(f"[{datetime.now()}] 发送通知: {message}")
        return {"status": "success", "message": message}

class PlaybookEngine:
    def __init__(self):
        self.steps = []

    def add_step(self, action, condition=None):
        self.steps.append({"action": action, "condition": condition})

    def run(self, context):
        for step in self.steps:
            if step["condition"]:
                if not step["condition"](context):
                    print(f"跳过步骤: {step['action'].__class__.__name__}")
                    continue
            result = step["action"].execute(context)
            context.update(result)
            time.sleep(0.5)
        return context

if __name__ == "__main__":
    engine = PlaybookEngine()
    # 定义条件：仅当IP为外网时封禁
    def is_external_ip(ctx):
        ip = ctx.get("ip", "")
        return not ip.startswith(("10.", "192.168.", "172.16."))
    engine.add_step(BlockIPAction(), condition=is_external_ip)
    engine.add_step(NotifyAction())
    context = {"ip": "185.130.5.133", "message": "检测到恶意IP，已自动封禁"}
    result = engine.run(context)
    print("执行完成:", result)
讲解：
动作封装具体操作，编排引擎按顺序执行。
条件分支实现动态决策（如只封禁公网IP）。
可扩展支持并行、循环、等待人工确认。
预期输出：打印封禁和通知动作，显示执行结果。

class FirewallManager:
    """模拟防火墙管理"""
    def __init__(self):
        self.blocked_ips = set()

    def block_ip(self, ip, duration=3600):
        """封堵IP，duration为秒"""
        if ip in self.blocked_ips:
            return {"status": "already_blocked", "ip": ip}
        # 模拟API调用
        print(f"封堵IP: {ip}，持续{duration}秒")
        self.blocked_ips.add(ip)
        return {"status": "success", "ip": ip, "duration": duration}

    def unblock_ip(self, ip):
        if ip in self.blocked_ips:
            self.blocked_ips.remove(ip)
            print(f"解封IP: {ip}")
            return {"status": "success", "ip": ip}
        return {"status": "not_found", "ip": ip}

"""
playbook_engine.py
基于YAML的剧本引擎
需要安装：pip install pyyaml
"""
import yaml
import importlib

class PlaybookRunner:
    def __init__(self, playbook_file):
        with open(playbook_file, 'r') as f:
            self.playbook = yaml.safe_load(f)
        self.actions = self._load_actions()

    def _load_actions(self):
        """动态加载动作模块"""
        actions = {}
        for act in self.playbook.get("actions", []):
            module_name = act["module"]
            class_name = act["class"]
            mod = importlib.import_module(module_name)
            cls = getattr(mod, class_name)
            actions[act["name"]] = cls()
        return actions

    def run(self, context):
        for step in self.playbook.get("steps", []):
            action_name = step["action"]
            params = step.get("params", {})
            # 将上下文中的变量替换到参数中
            for k, v in params.items():
                if isinstance(v, str) and v.startswith("{{") and v.endswith("}}"):
                    key = v[2:-2].strip()
                    params[k] = context.get(key, v)
            action = self.actions.get(action_name)
            if action:
                result = action.execute(params)
                context.update(result)
            else:
                print(f"动作 {action_name} 未找到")
        return context

# 示例动作模块（应与playbook同目录或可导入）

"""
log_analyzer.py
应急响应日志分析系统
功能：多源日志解析、异常检测、攻击时间线构建
"""

import re
import os
import gzip
import json
import time
from datetime import datetime
from collections import defaultdict, Counter
from typing import List, Dict, Any, Optional

class SecurityEvent:
    """安全事件类"""
    def __init__(self, timestamp: datetime, source: str, event_type: str, 
                 src_ip: str = None, dst_ip: str = None, detail: str = None):
        self.timestamp = timestamp
        self.source = source
        self.event_type = event_type
        self.src_ip = src_ip
        self.dst_ip = dst_ip
        self.detail = detail
    
    def to_dict(self) -> Dict:
        return {
            'timestamp': self.timestamp.isoformat(),
            'source': self.source,
            'type': self.event_type,
            'src_ip': self.src_ip,
            'dst_ip': self.dst_ip,
            'detail': self.detail
        }

class LogParser:
    """日志解析器基类"""
    
    @staticmethod
    def parse_apache(log_line: str) -> Optional[SecurityEvent]:
        """解析Apache访问日志"""
        pattern = re.compile(
            r'(?P<ip>\d+\.\d+\.\d+\.\d+) - - \[(?P<time>[^\]]+)\] '
            r'"(?P<method>\w+) (?P<path>[^\s]+) [^"]*" (?P<status>\d+) \d+'
        )
        match = pattern.search(log_line)
        if match:
            # 解析时间
            time_str = match.group('time')
            dt = datetime.strptime(time_str.split()[0], "%d/%b/%Y:%H:%M:%S")
            return SecurityEvent(
                timestamp=dt,
                source='Apache',
                event_type='web_request',
                src_ip=match.group('ip'),
                detail=f"{match.group('method')} {match.group('path')} {match.group('status')}"
            )
        return None
    
    @staticmethod
    def parse_auth(log_line: str) -> Optional[SecurityEvent]:
        """解析认证日志"""
        # 失败登录
        failed_pattern = re.compile(
            r'Failed password for .* from (?P<ip>\d+\.\d+\.\d+\.\d+)'
        )
        match = failed_pattern.search(log_line)
        if match:
            return SecurityEvent(
                timestamp=datetime.now(),
                source='Auth',
                event_type='auth_failure',
                src_ip=match.group('ip'),
                detail='密码错误'
            )
        
        # 成功登录
        success_pattern = re.compile(
            r'Accepted password for (?P<user>\w+) from (?P<ip>\d+\.\d+\.\d+\.\d+)'
        )
        match = success_pattern.search(log_line)
        if match:
            return SecurityEvent(
                timestamp=datetime.now(),
                source='Auth',
                event_type='auth_success',
                src_ip=match.group('ip'),
                detail=f"用户: {match.group('user')}"
            )
        return None

class AnomalyDetector:
    """异常检测器"""
    
    def __init__(self):
        self.ip_stats = defaultdict(Counter)
        self.ip_timeline = defaultdict(list)
    
    def detect_bruteforce(self, events: List[SecurityEvent], 
                          threshold: int = 10, time_window: int = 60) -> List[Dict]:
        """检测暴力破解攻击"""
        anomalies = []
        ip_failures = defaultdict(list)
        
        for event in events:
            if event.event_type == 'auth_failure' and event.src_ip:
                ip_failures[event.src_ip].append(event.timestamp)
        
        for ip, timestamps in ip_failures.items():
            if len(timestamps) >= threshold:
                # 检查时间窗口内次数
                recent = [t for t in timestamps 
                         if (datetime.now() - t).total_seconds() < time_window]
                if len(recent) >= threshold:
                    anomalies.append({
                        'type': 'bruteforce',
                        'ip': ip,
                        'count': len(timestamps),
                        'severity': 'high'
                    })
        return anomalies
    
    def detect_web_scan(self, events: List[SecurityEvent], 
                        threshold: int = 20) -> List[Dict]:
        """检测Web扫描行为"""
        ip_404 = defaultdict(int)
        for event in events:
            if event.event_type == 'web_request' and '404' in (event.detail or ''):
                if event.src_ip:
                    ip_404[event.src_ip] += 1
        
        anomalies = []
        for ip, count in ip_404.items():
            if count > threshold:
                anomalies.append({
                    'type': 'web_scan',
                    'ip': ip,
                    'count': count,
                    'severity': 'medium'
                })
        return anomalies

class TimelineBuilder:
    """攻击时间线构建器"""
    
    def __init__(self):
        self.events: List[SecurityEvent] = []
    
    def add_event(self, event: SecurityEvent):
        self.events.append(event)
    
    def build_timeline(self) -> List[SecurityEvent]:
        """按时间排序构建时间线"""
        return sorted(self.events, key=lambda x: x.timestamp)
    
    def export_html(self, filename: str = 'timeline.html'):
        """导出HTML格式时间线"""
        timeline = self.build_timeline()
        
        html = '''<!DOCTYPE html>
        <html>
        <head>
            <title>应急响应时间线</title>
            <style>
                body { font-family: monospace; margin: 20px; background: #1e1e1e; color: #d4d4d4; }
                .event { border-left: 3px solid; padding: 10px; margin: 10px 0; }
                .critical { border-left-color: #f48771; }
                .warning { border-left-color: #dcdcaa; }
                .info { border-left-color: #4ec9b0; }
                .time { color: #6a9955; font-size: 0.85em; }
                .type { color: #ce9178; font-weight: bold; }
            </style>
        </head>
        <body>
            <h1>应急响应事件时间线</h1>
        '''
        
        for event in timeline:
            severity_class = 'info'
            if event.event_type in ['auth_success']:
                severity_class = 'critical'
            elif 'fail' in event.event_type:
                severity_class = 'warning'
            
            html += f'''

                <div>详情: {event.detail or 'N/A'}</div>

        html += '</body></html>'

class LogAnalyzer:
    """日志分析器主类"""
    
    def __init__(self):
        self.parser = LogParser()
        self.detector = AnomalyDetector()
        self.timeline = TimelineBuilder()
        self.events: List[SecurityEvent] = []

class ProcessAnalyzer:
    """进程分析器"""
    
    # 可疑进程特征
    SUSPICIOUS_PROCESSES = {
        'minerd', 'xmrig', 'cgminer', 'bfgminer',  # 挖矿
        'nc', 'netcat', 'ncat', 'socat',            # 后门
        'msfconsole', 'meterpreter', 'beacon',      # 渗透
        'powershell -e', 'cmd.exe /c',              # 可疑命令
        'wget', 'curl',                             # 下载器
    }
    
    # 系统关键进程
    SYSTEM_PROCESSES = {'systemd', 'init', 'kernel', 'swapper'}
    
    def __init__(self):
        self.processes = []
        self.suspicious = []
        self.hidden_processes = []
    
    def get_all_processes(self) -> List[Dict]:
        """获取所有进程信息"""
        processes = []
        for proc in psutil.process_iter(['pid', 'name', 'exe', 'cmdline', 
                                         'username', 'cpu_percent', 'memory_percent']):
            try:
                info = proc.info
                info['pid'] = proc.pid
                processes.append(info)
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                continue
        return processes
    
    def detect_suspicious(self, processes: List[Dict]) -> List[Dict]:
        """检测可疑进程"""
        suspicious = []
        for proc in processes:
            name = proc.get('name', '').lower()
            cmdline = ' '.join(proc.get('cmdline', [])).lower()
            
            for pattern in self.SUSPICIOUS_PROCESSES:
                if pattern in name or pattern in cmdline:
                    suspicious.append({
                        'pid': proc['pid'],
                        'name': proc.get('name'),
                        'cmdline': cmdline[:200],
                        'username': proc.get('username'),
                        'reason': f'匹配特征: {pattern}',
                        'severity': 'high'
                    })
                    break
            
            # 检查异常CPU/内存使用
            if proc.get('cpu_percent', 0) > 80:
                suspicious.append({
                    'pid': proc['pid'],
                    'name': proc.get('name'),
                    'reason': f'CPU使用率异常: {proc["cpu_percent"]}%',
                    'severity': 'medium'
                })
            elif proc.get('memory_percent', 0) > 50:
                suspicious.append({
                    'pid': proc['pid'],
                    'name': proc.get('name'),
                    'reason': f'内存使用率异常: {proc["memory_percent"]}%',
                    'severity': 'medium'
                })
        
        return suspicious
    
    def check_hidden_processes(self) -> List[int]:
        """检测隐藏进程（通过/proc扫描）"""
        hidden = []
        if os.path.exists('/proc'):
            proc_dirs = set()
            for d in os.listdir('/proc'):
                if d.isdigit():
                    proc_dirs.add(int(d))
            
            running_pids = {p.pid for p in psutil.process_iter()}
            hidden = proc_dirs - running_pids
        return list(hidden)
    
    def get_network_connections(self, pid: Optional[int] = None) -> List[Dict]:
        """获取进程网络连接"""
        connections = []
        for conn in psutil.net_connections():
            if pid and conn.pid != pid:
                continue
            connections.append({
                'pid': conn.pid,
                'fd': conn.fd,
                'family': conn.family,
                'type': conn.type,
                'laddr': f"{conn.laddr.ip}:{conn.laddr.port}" if conn.laddr else None,
                'raddr': f"{conn.raddr.ip}:{conn.raddr.port}" if conn.raddr else None,
                'status': conn.status
            })
        return connections

class FileIntegrityMonitor:
    """文件完整性监控器"""
    
    def __init__(self, watch_dirs: List[str]):
        self.watch_dirs = watch_dirs
        self.baseline = {}
        self.db_file = 'file_integrity.json'
    
    def calculate_hash(self, filepath: str) -> Optional[str]:
        """计算文件哈希"""
        try:
            with open(filepath, 'rb') as f:
                return hashlib.sha256(f.read()).hexdigest()
        except (IOError, OSError):
            return None
    
    def scan_directory(self, directory: str) -> Dict:
        """扫描目录文件"""
        results = {}
        for root, dirs, files in os.walk(directory):
            # 跳过隐藏目录和特殊目录
            dirs[:] = [d for d in dirs if not d.startswith('.')]
            
            for file in files:
                filepath = os.path.join(root, file)
                file_hash = self.calculate_hash(filepath)
                if file_hash:
                    results[filepath] = {
                        'hash': file_hash,
                        'size': os.path.getsize(filepath),
                        'mtime': os.path.getmtime(filepath)
                    }
        return results
    
    def create_baseline(self):
        """创建基准线"""
        baseline = {}
        for directory in self.watch_dirs:
            if os.path.exists(directory):
                baseline.update(self.scan_directory(directory))
        
        self.baseline = baseline
        self.save_baseline()
        print(f"基准线创建完成，共 {len(baseline)} 个文件")
    
    def save_baseline(self):
        """保存基准线"""
        with open(self.db_file, 'w') as f:
            json.dump(self.baseline, f, indent=2)
    
    def load_baseline(self) -> Dict:
        """加载基准线"""
        if os.path.exists(self.db_file):
            with open(self.db_file, 'r') as f:
                self.baseline = json.load(f)
        return self.baseline
    
    def check_integrity(self) -> List[Dict]:
        """检查文件完整性"""
        changes = []
        current = {}
        
        for directory in self.watch_dirs:
            if os.path.exists(directory):
                current.update(self.scan_directory(directory))
        
        # 检测新增
        for path, info in current.items():
            if path not in self.baseline:
                changes.append({
                    'type': 'added',
                    'path': path,
                    'hash': info['hash'],
                    'size': info['size']
                })
        
        # 检测修改
        for path, baseline_info in self.baseline.items():
            current_info = current.get(path)
            if not current_info:
                changes.append({
                    'type': 'deleted',
                    'path': path
                })
            elif baseline_info['hash'] != current_info['hash']:
                changes.append({
                    'type': 'modified',
                    'path': path,
                    'old_hash': baseline_info['hash'],
                    'new_hash': current_info['hash']
                })
        
        return changes

"""
memory_forensics.py
内存取证分析系统
功能：内存镜像解析、进程内存分析、恶意代码检测
"""

import os
import sys
import struct
import hashlib
from datetime import datetime
from typing import List, Dict, Optional, Tuple

class ProcessMemoryAnalyzer:
    """进程内存分析器"""
    
    def __init__(self, pid: int):
        self.pid = pid
        self.maps_file = f'/proc/{pid}/maps'
        self.mem_file = f'/proc/{pid}/mem'
    
    def read_maps(self) -> List[Dict]:
        """读取内存映射"""
        regions = []
        
        if not os.path.exists(self.maps_file):
            return regions
        
        with open(self.maps_file, 'r') as f:
            for line in f:
                parts = line.strip().split()
                if len(parts) >= 5:
                    addr_range, perms, offset, dev, inode = parts[:5]
                    start, end = [int(x, 16) for x in addr_range.split('-')]
                    
                    regions.append({
                        'start': start,
                        'end': end,
                        'size': end - start,
                        'perms': perms,
                        'path': parts[-1] if len(parts) > 5 else ''
                    })
        return regions
    
    def dump_region(self, start: int, end: int, output_file: str) -> int:
        """转储内存区域"""
        size = end - start
        try:
            with open(self.mem_file, 'rb') as mem:
                mem.seek(start)
                data = mem.read(size)
                
                with open(output_file, 'wb') as out:
                    out.write(data)
                return len(data)
        except (IOError, OSError):
            return 0
    
    def scan_for_strings(self, min_length: int = 4) -> List[str]:
        """扫描内存中的字符串"""
        strings = []
        regions = [r for r in self.read_maps() if 'r' in r['perms']]
        
        for region in regions[:10]:  # 限制扫描区域
            try:
                with open(self.mem_file, 'rb') as mem:
                    mem.seek(region['start'])
                    data = mem.read(min(region['size'], 1024 * 1024))
                    
                    # 提取可打印字符串
                    current = []
                    for b in data:
                        if 32 <= b <= 126:
                            current.append(chr(b))
                        else:
                            if len(current) >= min_length:
                                strings.append(''.join(current))
                            current = []
            except:
                continue
        
        return strings
    
    def scan_for_pe(self) -> List[Tuple[int, str]]:
        """扫描PE/DLL文件头"""
        pe_signatures = []
        regions = [r for r in self.read_maps() if 'r' in r['perms']]
        
        # PE文件头：MZ
        for region in regions:
            try:
                with open(self.mem_file, 'rb') as mem:
                    mem.seek(region['start'])
                    data = mem.read(1024)
                    
                    if data.startswith(b'MZ'):
                        pe_signatures.append((region['start'], 'PE/EXE'))
                    elif data.startswith(b'\x7fELF'):
                        pe_signatures.append((region['start'], 'ELF'))
            except:
                continue
        
        return pe_signatures

class MemoryForensics:
    """内存取证分析器"""
    
    def __init__(self, memory_dump: Optional[str] = None):
        self.memory_dump = memory_dump
        self.findings = {
            'timestamp': datetime.now().isoformat(),
            'processes': [],
            'strings': [],
            'pe_signatures': [],
            'suspicious': []
        }
    
    def analyze_running_system(self):
        """分析运行中的系统"""
        import psutil
        
        for proc in psutil.process_iter(['pid', 'name', 'exe', 'cmdline']):
            try:
                pid = proc.info['pid']
                name = proc.info.get('name', 'Unknown')
                
                analyzer = ProcessMemoryAnalyzer(pid)
                
                # 扫描可疑字符串
                strings = analyzer.scan_for_strings()
                suspicious_strings = [s for s in strings if any(
                    keyword in s.lower() for keyword in 
                    ['password', 'secret', 'key', 'token', 'admin']
                )]
                
                # 扫描PE签名
                pe_sigs = analyzer.scan_for_pe()
                
                if suspicious_strings or pe_sigs:
                    self.findings['processes'].append({
                        'pid': pid,
                        'name': name,
                        'cmdline': ' '.join(proc.info.get('cmdline', [])),
                        'suspicious_strings': suspicious_strings[:10],
                        'pe_signatures': pe_sigs
                    })
            except:
                continue
    
    def detect_hidden_modules(self) -> List[Dict]:
        """检测隐藏模块"""
        hidden = []
        
        # 检查LD_PRELOAD劫持
        ld_preload = os.environ.get('LD_PRELOAD', '')
        if ld_preload:
            hidden.append({
                'type': 'LD_PRELOAD',
                'value': ld_preload,
                'severity': 'high'
            })
        
        # 检查动态链接器配置
        if os.path.exists('/etc/ld.so.preload'):
            with open('/etc/ld.so.preload', 'r') as f:
                content = f.read().strip()
                if content:
                    hidden.append({
                        'type': 'ld.so.preload',
                        'value': content,
                        'severity': 'high'
                    })
        
        return hidden
    
    def extract_network_artifacts(self) -> List[Dict]:
        """提取网络痕迹"""
        artifacts = []
        
        # 检查/proc/net/下的连接信息
        net_files = ['tcp', 'udp', 'tcp6', 'udp6']
        
        for net_file in net_files:
            path = f'/proc/net/{net_file}'
            if os.path.exists(path):
                with open(path, 'r') as f:
                    lines = f.readlines()[1:]  # 跳过标题行
                    
                    for line in lines:
                        parts = line.split()
                        if len(parts) >= 4:
                            # 解析本地和远程地址
                            local = parts[1]
                            remote = parts[2]
                            state = parts[3]
                            
                            artifacts.append({
                                'protocol': net_file,
                                'local': local,
                                'remote': remote,
                                'state': state
                            })
        
        return artifacts
    
    def yara_scan(self, rules_path: str) -> List[Dict]:
        """YARA规则扫描（需要yara-python）"""
        matches = []
        
        try:
            import yara
            rules = yara.compile(rules_path)
            
            # 扫描内存中的进程
            import psutil
            for proc in psutil.process_iter(['pid', 'name']):
                try:
                    matches.extend(rules.match(proc.pid))
                except:
                    continue
        except ImportError:
            print("YARA支持未安装")
        
        return matches
    
    def generate_report(self) -> str:
        """生成取证报告"""
        report = f"""
        ========================================
        内存取证分析报告
        ========================================
        
        分析时间: {self.findings['timestamp']}
        
        【可疑进程】
        """
        
        for proc in self.findings['processes'][:10]:
            report += f"""
        PID: {proc['pid']}
        名称: {proc['name']}
        命令行: {proc['cmdline'][:100]}
        """
        
        if self.findings['suspicious']:
            report += "\n【可疑发现】\n"
            for sus in self.findings['suspicious']:
                report += f"  - {sus}\n"
        
        report += """
        【建议措施】
        1. 对可疑进程进行内存转储分析
        2. 检查进程对应的可执行文件
        3. 分析网络连接的目标地址
        4. 使用YARA规则进行深度扫描
        """
        
        return report

import socket
import struct
from datetime import datetime
from collections import defaultdict, Counter
from typing import List, Dict, Optional, Tuple
import hashlib

class PacketParser:
    """数据包解析器"""
    
    @staticmethod
    def ethernet_header(data: bytes) -> Tuple[str, str, int]:
        """解析以太网头"""
        dest_mac = ':'.join(f'{b:02x}' for b in data[0:6])
        src_mac = ':'.join(f'{b:02x}' for b in data[6:12])
        eth_type = struct.unpack('!H', data[12:14])[0]
        return dest_mac, src_mac, eth_type
    
    @staticmethod
    def ip_header(data: bytes) -> Dict:
        """解析IP头"""
        version_ihl = data[0]
        version = version_ihl >> 4
        ihl = version_ihl & 0xF
        
        tos = data[1]
        total_length = struct.unpack('!H', data[2:4])[0]
        identification = struct.unpack('!H', data[4:6])[0]
        flags_offset = struct.unpack('!H', data[6:8])[0]
        ttl = data[8]
        protocol = data[9]
        checksum = struct.unpack('!H', data[10:12])[0]
        src_ip = socket.inet_ntoa(data[12:16])
        dst_ip = socket.inet_ntoa(data[16:20])
        
        return {
            'version': version,
            'ihl': ihl,
            'tos': tos,
            'total_length': total_length,
            'identification': identification,
            'ttl': ttl,
            'protocol': protocol,
            'src_ip': src_ip,
            'dst_ip': dst_ip
        }
    
    @staticmethod
    def tcp_header(data: bytes) -> Dict:
        """解析TCP头"""
        src_port = struct.unpack('!H', data[0:2])[0]
        dst_port = struct.unpack('!H', data[2:4])[0]
        seq_num = struct.unpack('!I', data[4:8])[0]
        ack_num = struct.unpack('!I', data[8:12])[0]
        
        offset_reserved_flags = struct.unpack('!H', data[12:14])[0]
        data_offset = (offset_reserved_flags >> 12) & 0xF
        flags = offset_reserved_flags & 0x1FF
        window = struct.unpack('!H', data[14:16])[0]
        checksum = struct.unpack('!H', data[16:18])[0]
        urgent = struct.unpack('!H', data[18:20])[0]
        
        return {
            'src_port': src_port,
            'dst_port': dst_port,
            'seq_num': seq_num,
            'ack_num': ack_num,
            'data_offset': data_offset,
            'flags': flags,
            'window': window,
            'checksum': checksum,
            'urgent': urgent
        }
    
    @staticmethod
    def udp_header(data: bytes) -> Dict:
        """解析UDP头"""
        src_port = struct.unpack('!H', data[0:2])[0]
        dst_port = struct.unpack('!H', data[2:4])[0]
        length = struct.unpack('!H', data[4:6])[0]
        checksum = struct.unpack('!H', data[6:8])[0]
        
        return {
            'src_port': src_port,
            'dst_port': dst_port,
            'length': length,
            'checksum': checksum
        }

class PCAPReader:
    """PCAP文件读取器"""
    
    def __init__(self, pcap_file: str):
        self.pcap_file = pcap_file
        self.packets = []
        self.parser = PacketParser()
    
    def read_pcap(self) -> List[Dict]:
        """读取PCAP文件"""
        packets = []
        
        with open(self.pcap_file, 'rb') as f:
            # 跳过PCAP头
            f.read(24)
            
            while True:
                # 读取包头
                header = f.read(16)
                if len(header) < 16:
                    break
                
                ts_sec, ts_usec, incl_len, orig_len = struct.unpack('IIII', header)
                timestamp = datetime.fromtimestamp(ts_sec + ts_usec / 1000000.0)
                
                # 读取数据包
                packet_data = f.read(incl_len)
                if len(packet_data) < incl_len:
                    break
                
                packet = self.parse_packet(packet_data, timestamp)
                if packet:
                    packets.append(packet)
        
        self.packets = packets
        return packets
    
    def parse_packet(self, data: bytes, timestamp: datetime) -> Optional[Dict]:
        """解析数据包"""
        if len(data) < 14:
            return None
        
        # 解析以太网头
        dest_mac, src_mac, eth_type = self.parser.ethernet_header(data)
        
        # 只处理IP包
        if eth_type != 0x0800:
            return None
        
        ip_start = 14
        
        # 解析IP头
        ip_header = self.parser.ip_header(data[ip_start:])
        ip_header_len = ip_header['ihl'] * 4
        
        transport_start = ip_start + ip_header_len
        protocol = ip_header['protocol']
        
        packet_info = {
            'timestamp': timestamp,
            'src_mac': src_mac,
            'dst_mac': dest_mac,
            'src_ip': ip_header['src_ip'],
            'dst_ip': ip_header['dst_ip'],
            'protocol': 'TCP' if protocol == 6 else 'UDP' if protocol == 17 else 'ICMP' if protocol == 1 else str(protocol),
            'length': ip_header['total_length']
        }
        
        # 解析传输层
        if protocol == 6 and len(data) > transport_start + 20:
            tcp = self.parser.tcp_header(data[transport_start:])
            packet_info['src_port'] = tcp['src_port']
            packet_info['dst_port'] = tcp['dst_port']
            packet_info['tcp_flags'] = tcp['flags']
        elif protocol == 17 and len(data) > transport_start + 8:
            udp = self.parser.udp_header(data[transport_start:])
            packet_info['src_port'] = udp['src_port']
            packet_info['dst_port'] = udp['dst_port']
        
        return packet_info

class TrafficAnalyzer:
    """流量分析器"""
    
    def __init__(self):
        self.flows = defaultdict(lambda: {'packets': [], 'bytes': 0})
    
    def analyze_packets(self, packets: List[Dict]):
        """分析数据包"""
        for packet in packets:
            flow_key = f"{packet['src_ip']}:{packet.get('src_port', '')}->{packet['dst_ip']}:{packet.get('dst_port', '')}"
            self.flows[flow_key]['packets'].append(packet)
            self.flows[flow_key]['bytes'] += packet.get('length', 0)
    
    def detect_port_scan(self) -> List[Dict]:
        """检测端口扫描"""
        scanners = defaultdict(set)
        
        for flow_key, flow in self.flows.items():
            src_ip = flow_key.split('->')[0].split(':')[0]
            # 提取目的端口
            dst_port = flow_key.split('->')[1].split(':')[1] if '->' in flow_key else None
            if dst_port:
                scanners[src_ip].add(int(dst_port))
        
        port_scans = []
        for ip, ports in scanners.items():
            if len(ports) > 20:  # 扫描超过20个端口
                port_scans.append({
                    'ip': ip,
                    'ports_scanned': len(ports),
                    'severity': 'medium'
                })
        
        return port_scans
    
    def detect_beacon(self, threshold: float = 0.1) -> List[Dict]:
        """检测Beacon通信（周期性请求）"""
        beacon_candidates = []
        
        for flow_key, flow in self.flows.items():
            if len(flow['packets']) < 3:
                continue
            
            # 提取时间间隔
            timestamps = [p['timestamp'] for p in flow['packets']]
            intervals = [(timestamps[i+1] - timestamps[i]).total_seconds() 
                        for i in range(len(timestamps)-1)]
            
            if intervals:
                avg_interval = sum(intervals) / len(intervals)
                variance = sum((i - avg_interval) ** 2 for i in intervals) / len(intervals)
                
                # 低方差表示周期性
                if variance < avg_interval * threshold:
                    beacon_candidates.append({
                        'flow': flow_key,
                        'avg_interval': avg_interval,
                        'variance': variance,
                        'packet_count': len(flow['packets'])
                    })
        
        return beacon_candidates

"""
malware_analyzer.py
恶意软件分析系统
功能：静态分析、动态行为监控、YARA扫描
"""

import os
import sys
import hashlib
import subprocess
import tempfile
import json
import time
from datetime import datetime
from typing import List, Dict, Optional, Tuple

class StaticAnalyzer:
    """静态分析器"""
    
    def __init__(self, filepath: str):
        self.filepath = filepath
        self.file_size = 0
        self.md5_hash = ""
        self.sha256_hash = ""
        self.strings = []
        self.sections = []
        self.imports = []
    
    def calculate_hashes(self) -> Dict[str, str]:
        """计算文件哈希"""
        hashes = {}
        
        with open(self.filepath, 'rb') as f:
            data = f.read()
            hashes['md5'] = hashlib.md5(data).hexdigest()
            hashes['sha1'] = hashlib.sha1(data).hexdigest()
            hashes['sha256'] = hashlib.sha256(data).hexdigest()
        
        self.md5_hash = hashes['md5']
        self.sha256_hash = hashes['sha256']
        return hashes
    
    def extract_strings(self, min_length: int = 4) -> List[str]:
        """提取字符串"""
        strings = []
        
        with open(self.filepath, 'rb') as f:
            data = f.read()
            current = []
            
            for byte in data:
                if 32 <= byte <= 126:
                    current.append(chr(byte))
                else:
                    if len(current) >= min_length:
                        strings.append(''.join(current))
                    current = []
        
        self.strings = strings
        return strings
    
    def detect_suspicious_strings(self) -> List[str]:
        """检测可疑字符串"""
        suspicious = []
        indicators = [
            'cmd.exe', 'powershell', 'wscript', 'cscript',
            'reg add', 'schtasks', 'sc config',
            'http://', 'https://', '.onion',
            'base64', 'AES', 'encrypt',
            'MZ', 'This program cannot be run in DOS mode'
        ]
        
        for s in self.strings:
            for indicator in indicators:
                if indicator.lower() in s.lower():
                    suspicious.append(s[:100])
                    break
        
        return suspicious
    
    def check_packer(self) -> Optional[str]:
        """检测加壳"""
        # 检查熵值（粗略）
        import math
        with open(self.filepath, 'rb') as f:
            data = f.read()
            
            if not data:
                return None
            
            # 计算熵
            freq = {}
            for byte in data:
                freq[byte] = freq.get(byte, 0) + 1
            
            entropy = 0
            for count in freq.values():
                p = count / len(data)
                entropy -= p * math.log2(p)
            
            # 高熵值表示可能加壳
            if entropy > 6.5:
                return f"高熵值 ({entropy:.2f})，可能加壳"
        
        return None
    
    def get_pe_info(self) -> Dict:
        """获取PE文件信息（仅Windows）"""
        info = {}
        
        if sys.platform == 'win32':
            try:
                import pefile
                pe = pefile.PE(self.filepath)
                
                info['entry_point'] = hex(pe.OPTIONAL_HEADER.AddressOfEntryPoint)
                info['image_base'] = hex(pe.OPTIONAL_HEADER.ImageBase)
                info['sections'] = [s.Name.decode().strip('\x00') for s in pe.sections]
            except ImportError:
                pass
        
        return info

class DynamicAnalyzer:
    """动态行为分析器"""
    
    def __init__(self, sample_path: str):
        self.sample_path = sample_path
        self.behavior = {
            'processes': [],
            'files': [],
            'registry': [],
            'network': []
        }
    
    def run_in_sandbox(self, timeout: int = 30) -> Dict:
        """在沙箱中运行样本"""
        # 创建临时目录
        temp_dir = tempfile.mkdtemp()
        
        # 监控子进程
        try:
            proc = subprocess.Popen(
                [self.sample_path],
                cwd=temp_dir,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                timeout=timeout
            )
            
            self.behavior['processes'].append({
                'pid': proc.pid,
                'cmdline': f"{self.sample_path}"
            })
            
            # 等待进程结束
            stdout, stderr = proc.communicate(timeout=timeout)
            self.behavior['files'] = self.scan_directory(temp_dir)
            
        except subprocess.TimeoutExpired:
            proc.kill()
            self.behavior['processes'].append({'note': '进程超时被终止'})
        except Exception as e:
            self.behavior['error'] = str(e)
        
        # 清理
        import shutil
        shutil.rmtree(temp_dir, ignore_errors=True)
        
        return self.behavior
    
    def scan_directory(self, directory: str) -> List[str]:
        """扫描目录中的新文件"""
        files = []
        for root, dirs, filenames in os.walk(directory):
            for filename in filenames:
                files.append(os.path.join(root, filename))
        return files

class YaraScanner:
    """YARA规则扫描器"""
    
    def __init__(self):
        self.rules = self.get_default_rules()
    
    def get_default_rules(self) -> str:
        """获取默认规则"""
        return """
        rule SuspiciousStrings {
            strings:
                $s1 = "cmd.exe" nocase
                $s2 = "powershell" nocase
                $s3 = "http://" nocase
                $s4 = "https://" nocase
                $s5 = "base64" nocase
            condition:
                2 of them
        }
        
        rule CryptoMiner {
            strings:
                $m1 = "stratum+tcp" nocase
                $m2 = "miner" nocase
                $m3 = "xmrig" nocase
            condition:
                any of them
        }
        """
    
    def scan(self, filepath: str) -> List[Dict]:
        """扫描文件"""
        matches = []
        
        try:
            import yara
            compiled_rules = yara.compile(source=self.rules)
            results = compiled_rules.match(filepath)
            
            for result in results:
                matches.append({
                    'rule': result.rule,
                    'tags': result.tags,
                    'strings': [(s.identifier, s.string) for s in result.strings]
                })
        except ImportError:
            # 模拟扫描结果
            if 'miner' in filepath.lower():
                matches.append({'rule': 'CryptoMiner', 'tags': [], 'strings': []})
        
        return matches

class MalwareAnalyzer:
    """恶意软件分析主类"""
    
    def __init__(self):
        self.static_analyzer = None
        self.dynamic_analyzer = None
        self.yara_scanner = YaraScanner()
        self.results = {}
    
    def analyze(self, filepath: str, dynamic: bool = False) -> Dict:
        """分析文件"""
        self.results = {
            'file': filepath,
            'timestamp': datetime.now().isoformat(),
            'static': {},
            'dynamic': {},
            'yara': []
        }
        
        # 静态分析
        self.static_analyzer = StaticAnalyzer(filepath)
        self.static_analyzer.calculate_hashes()
        self.static_analyzer.extract_strings()
        
        self.results['static'] = {
            'size': os.path.getsize(filepath),
            'md5': self.static_analyzer.md5_hash,
            'sha256': self.static_analyzer.sha256_hash,
            'suspicious_strings': self.static_analyzer.detect_suspicious_strings()[:20],
            'packer': self.static_analyzer.check_packer()
        }
        
        # YARA扫描
        self.results['yara'] = self.yara_scanner.scan(filepath)
        
        # 动态分析（可选）
        if dynamic:
            self.dynamic_analyzer = DynamicAnalyzer(filepath)
            self.results['dynamic'] = self.dynamic_analyzer.run_in_sandbox()
        
        # 计算风险评分
        self.results['risk_score'] = self.calculate_risk_score()
        
        return self.results
    
    def calculate_risk_score(self) -> int:
        """计算风险评分"""
        score = 0
        
        # 可疑字符串加分
        suspicious_count = len(self.results['static'].get('suspicious_strings', []))
        score += min(suspicious_count * 5, 40)
        
        # 加壳加分
        if self.results['static'].get('packer'):
            score += 20
        
        # YARA匹配加分
        score += len(self.results.get('yara', [])) * 15
        
        return min(score, 100)
    
    def generate_report(self) -> str:
        """生成报告"""
        report = f"""
        ========================================
        恶意软件分析报告
        ========================================
        
        文件: {self.results['file']}
        分析时间: {self.results['timestamp']}
        风险评分: {self.results['risk_score']}/100
        
        【静态分析】
        - 文件大小: {self.results['static']['size']} bytes
        - MD5: {self.results['static']['md5']}
        - SHA256: {self.results['static']['sha256']}
        - 加壳检测: {self.results['static']['packer'] or '未检测到'}
        
        【可疑字符串】
        """
        
        for s in self.results['static'].get('suspicious_strings', [])[:10]:
            report += f"  - {s[:80]}\n"
        
        if self.results['yara']:
            report += "\n【YARA规则匹配】\n"
            for match in self.results['yara']:
                report += f"  - 规则: {match['rule']}\n"
        
        return report
